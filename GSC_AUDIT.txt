class GSCReadOnlyAdmin(GSCAdmin):

    def get_readonly_fields(self, request, obj=None):
        return [
            field.name
            for field in self.model._meta.fields
        ]

    def has_add_permission(self, request):
        return False

    def has_change_permission(self, request, obj=None):
        return super().has_view_permission(request, obj)

    class ModeReception(models.TextChoices):

    reference = models.CharField(

    sens = models.CharField(

    date_courrier = models.DateField(default=timezone.localdate)

    # Ancien champ conservé pour compatibilité.

    date_reception_physique = models.DateTimeField(

    )

    nom_livreur = models.CharField(max_length=180, blank=True)

    date_expedition = models.DateField(
        null=True,
        blank=True
    )

    expediteur = models.CharField(max_length=180)
    destinataire = models.CharField(max_length=180)
    objet = models.CharField(max_length=250)
    description = models.TextField(blank=True)

    categorie = models.ForeignKey(

    )

    service = models.ForeignKey(

        related_name="courriers",
        null=True,
        blank=True
    )

    responsable = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="courriers_responsables",
        null=True,
        blank=True
    )

    priorite = models.CharField(
        max_length=20,
        choices=Priorite.choices,
        default=Priorite.NORMALE
    )

    confidentialite = models.CharField(
        max_length=20,
        choices=Confidentialite.choices,
        default=Confidentialite.NORMALE
    )

    statut = models.CharField(
        max_length=20,
        choices=Statut.choices,
        default=Statut.ENREGISTRE
    )

    date_limite = models.DateField(null=True, blank=True)
    date_cloture = models.DateField(null=True, blank=True)

    document = models.FileField(
        upload_to="courriers/%Y/%m/",
        blank=True,
        null=True
    )

    courrier_origine = models.ForeignKey(
        "self",
        on_delete=models.PROTECT,
        related_name="courriers_reponse",
        null=True,
        blank=True
    )

    cree_par = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="courriers_crees"
    )

    cree_le = models.DateTimeField(auto_now_add=True)
    modifie_le = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-cree_le"]
        verbose_name = "Courrier"
        verbose_name_plural = "Courriers"

        permissions = [
            ("transmettre_courrier", "Peut transmettre un courrier"),
            ("cloturer_courrier", "Peut clôturer un courrier"),
            ("traiter_courrier", "Peut traiter un courrier"),
            ("retourner_courrier", "Peut retourner un courrier"),
            ("reouvrir_courrier", "Peut rouvrir un courrier"),
        ]

    def save(self, *args, **kwargs):
        if not self.reference:
            prefix = (
                "ENT"
                if self.sens == self.Sens.ENTRANT
                else "SOR"
            )

            annee = timezone.localdate().year

            with transaction.atomic():
                compteur, cree = (
                    CompteurReference.objects
                    .select_for_update()
                    .get_or_create(
                        sens=self.sens,
                        annee=annee,
                        defaults={"dernier_numero": 0}
                    )
                )

                if cree or compteur.dernier_numero == 0:
                    anciennes_references = (
                        Courrier.objects
                        .filter(
                            reference__startswith=f"{prefix}-{annee}-"
                        )
                        .values_list("reference", flat=True)
                    )

                    maximum = 0

                    for ancienne in anciennes_references:
                        try:
                            numero = int(ancienne.rsplit("-", 1)[1])
                            maximum = max(maximum, numero)
                        except (ValueError, IndexError):
                            continue

                    compteur.dernier_numero = max(
                        compteur.dernier_numero,
                        maximum
                    )

                compteur.dernier_numero += 1
                compteur.save(update_fields=["dernier_numero"])

                self.reference = (
                    f"{prefix}-{annee}-"
                    f"{compteur.dernier_numero:06d}"
                )

                super().save(*args, **kwargs)
        else:
            super().save(*args, **kwargs)

    def __str__(self):
        return f"{self.reference} – {self.objet}"

class MultipleFileField(forms.FileField):
    widget = MultipleFileInput

    def clean(self, data, initial=None):
        if not data:
            return []

        fichiers = data if isinstance(data, (list, tuple)) else [data]

        resultats = []

        for fichier in fichiers:
            fichier_nettoye = super().clean(fichier, initial)

            if fichier_nettoye:
                resultats.append(fichier_nettoye)

        return resultats


# =====================================================
# VALIDATION COMMUNE DES FICHIERS
# =====================================================

def valider_fichier(fichier):
    extension = Path(fichier.name).suffix.lower()

    if extension not in EXTENSIONS_AUTORISEES:
        raise forms.ValidationError(
            f"Le format du fichier {fichier.name} n'est pas autorisé."
        )

    if fichier.size > TAILLE_MAX_FICHIER:
        raise forms.ValidationError(
            f"Le fichier {fichier.name} dépasse la limite de 20 Mo."
        )

    return fichier

# =====================================================
# NORMALISATION DES GROUPES
# =====================================================

def normaliser_nom(valeur):

    valeur = unicodedata.normalize("NFD", valeur or "")

    valeur = "".join(
        caractere
        for caractere in valeur
        if unicodedata.category(caractere) != "Mn"
    )

    return valeur.strip().casefold()


# =====================================================
# IDENTIFICATION DU PROFIL
# =====================================================

def obtenir_profil(user):

    if not user.is_authenticated or not user.is_active:
        return None

    if user.is_superuser:
        return "ADMIN"

    groupes = {
        normaliser_nom(nom)
        for nom in user.groups.values_list(
            "name",
            flat=True
        )
    }

    correspondances = [
        ("accueil", "ACCUEIL"),
        ("secretariat", "SECRETARIAT"),
        ("direction", "DIRECTION"),
        ("responsable de service", "RESPONSABLE"),
        ("agent", "AGENT"),
    ]

    for nom, profil in correspondances:

        if nom in groupes:
            return profil

    return None

def courrier_est_confidentiel(courrier):

    return (
        courrier.confidentialite in [
            Courrier.Confidentialite.CONFIDENTIELLE,
            Courrier.Confidentialite.STRICTE,
        ]
        or courrier.priorite == Courrier.Priorite.CONFIDENTIELLE
    )

    return (
        courrier.statut != Courrier.Statut.ENREGISTRE
        or courrier.affectations.filter(
            active=True
        ).exists()
    )

        return False

    maintenant = timezone.now()

    return DelegationTransmission.objects.filter(

# =====================================================
# CONSULTATION D'UN COURRIER
# =====================================================

def utilisateur_peut_consulter_courrier(user, courrier):

    profil = obtenir_profil(user)

    if profil is None or courrier is None:
        return False

    if not utilisateur_peut_consulter(user):
        return False

    # Administrateur et Direction : consultation globale.
    if profil in ["ADMIN", "DIRECTION"]:
        return True

    # Accueil et Secrétariat.
    if profil in ["ACCUEIL", "SECRETARIAT"]:

        if courrier_est_confidentiel(courrier):

            return not courrier_est_deja_transmis(
                courrier
            )

        return True

    # Responsable : uniquement son service.
    if profil == "RESPONSABLE":

        return utilisateur_est_responsable_du_courrier(
            user,
            courrier
        )

    # Agent : uniquement ses affectations actives,
    # jamais les courriers confidentiels.
    if profil == "AGENT":

        if courrier_est_confidentiel(courrier):
            return False

        return courrier.affectations.filter(
            agent=user,
            active=True
        ).exists()

    return False


# =====================================================
# MODIFICATION D'UN COURRIER
# =====================================================

def utilisateur_peut_modifier_courrier(user, courrier):

    profil = obtenir_profil(user)

    if profil is None or courrier is None:
        return False

    # Accueil, Direction et agents ne modifient pas.
    if profil in ["ACCUEIL", "DIRECTION", "AGENT"]:
        return False

    # Administrateur : modification selon permission Django.
    if profil == "ADMIN":

        return user.has_perm(
            "courrier.change_courrier"
        )

    if not utilisateur_peut_consulter_courrier(
        user,
        courrier
    ):
        return False

    # Secrétariat : uniquement avant transmission.
    if profil == "SECRETARIAT":

        return (
            user.has_perm("courrier.change_courrier")
            and not courrier_est_deja_transmis(courrier)
        )

    # Responsable : fiche de son service uniquement.
    # Le changement de service est interdit dans la vue.
    if profil == "RESPONSABLE":

        return (
            utilisateur_est_responsable_du_courrier(
                user,
                courrier
            )
            and user.has_perm("courrier.change_courrier")
            and courrier.statut != Courrier.Statut.CLOTURE
        )

    return False


# =====================================================
# TRANSMISSION INITIALE
# =====================================================

def utilisateur_peut_transmettre(user, courrier=None):

    profil = obtenir_profil(user)

    if profil is None:
        return False

    if profil == "ADMIN":
        return True

    if profil not in ["ACCUEIL", "SECRETARIAT"]:
        return False

    if not user.has_perm(
        "courrier.transmettre_courrier"
    ):
        return False

    if courrier is not None:

        if courrier.sens != Courrier.Sens.ENTRANT:
            return False

        if courrier.statut != Courrier.Statut.ENREGISTRE:
            return False

        if courrier_est_deja_transmis(courrier):
            return False

    # Accueil : transmission normale.
    if profil == "ACCUEIL":
        return True

    # Secrétariat : remplacement temporaire autorisé.
    return utilisateur_a_delegation_valide(user)


# =====================================================
# AFFECTATION A UN AGENT
# =====================================================

def utilisateur_peut_affecter_agent(user, courrier):

    profil = obtenir_profil(user)

    if profil == "ADMIN":
        return True

    if profil != "RESPONSABLE" or courrier is None:
        return False

    return (
        utilisateur_est_responsable_du_courrier(
            user,
            courrier
        )
        and user.has_perm("courrier.change_courrier")

# =====================================================

def utilisateur_peut_traiter_courrier(user, courrier):

    profil = obtenir_profil(user)

    if profil not in ["AGENT", "RESPONSABLE"]:
        return False

    if courrier is None:
        return False

    if courrier.sens != Courrier.Sens.ENTRANT:
        return False

    if courrier.statut not in [
        Courrier.Statut.TRANSMIS,
        Courrier.Statut.EN_COURS,
        Courrier.Statut.EN_ATTENTE,
    ]:
        return False

    if not utilisateur_peut_consulter_courrier(
        user,
        courrier
    ):
        return False

    if not user.has_perm(
        "courrier.traiter_courrier"
    ):
        return False

    if profil == "AGENT":

        return courrier.affectations.filter(
            agent=user,
            active=True
        ).exists()

    return utilisateur_est_responsable_du_courrier(
        user,
        courrier
    )

            ".pdf", ".doc", ".docx",
            ".xls", ".xlsx",
            ".jpg", ".jpeg", ".png"
        ]

        extension = Path(fichier.name).suffix.lower()

            raise forms.ValidationError(

            raise forms.ValidationError(
                "Le fichier ne doit pas dépasser 20 Mo."
            )

        return fichier

def courriers_accessibles(user):
    profil = verifier_droit_consultation(user)

    qs = Courrier.objects.select_related(
        "service",
        "responsable",
        "cree_par"
    ).distinct()

    if profil in ["ADMIN", "DIRECTION"]:
        return qs

    confidentialite_normale = (
        Q(confidentialite=Courrier.Confidentialite.NORMALE)
        & ~Q(priorite=Courrier.Priorite.CONFIDENTIELLE)
    )

    confidentialite_restraint = (
        Q(confidentialite__in=[
            Courrier.Confidentialite.CONFIDENTIELLE,
            Courrier.Confidentialite.STRICTE,
        ])
        | Q(priorite=Courrier.Priorite.CONFIDENTIELLE)
    )

    if profil in ["ACCUEIL", "SECRETARIAT"]:
        courriers_normaux = qs.filter(
            confidentialite_normale
        )

        courriers_confidentiels_non_transmis = qs.filter(
            confidentialite_restraint,
            statut="ENREGISTRE"
        ).exclude(
            affectations__active=True
        )

        return (
            courriers_normaux
            | courriers_confidentiels_non_transmis
        ).distinct()

    if profil == "RESPONSABLE":
        return qs.filter(
            service__responsable=user
        ).distinct()

    if profil == "AGENT":
        return qs.filter(
            affectations__agent=user,
            affectations__active=True
        ).filter(
            confidentialite_normale
        ).distinct()

    return qs.none()


def obtenir_courrier_accessible(user, pk):
    return get_object_or_404(
        courriers_accessibles(user),
        pk=pk
    )

@login_required
def dashboard(request):
    profil = obtenir_profil(request.user)
    qs = courriers_accessibles(request.user)
    aujourd_hui = timezone.localdate()

    context = {
        "profil": profil,
        "total": qs.count(),
        "entrants": qs.filter(sens="ENTRANT").count(),
        "sortants": qs.filter(sens="SORTANT").count(),
        "en_cours": qs.filter(
            statut__in=["TRANSMIS", "EN_COURS", "EN_ATTENTE"]
        ).count(),
        "traites": qs.filter(
            statut__in=["TRAITE", "CLOTURE"]
        ).count(),
        "en_retard": qs.filter(
            date_limite__lt=aujourd_hui
        ).exclude(
            statut__in=["TRAITE", "CLOTURE"]
        ).count(),
        "recents": qs.order_by("-cree_le")[:8],
        "a_transmettre": 0,
        "a_traiter": 0,
        "a_cloturer": 0,
        "retours_en_attente": 0,
        "mes_affectations": 0,
        "mes_services": 0,
    }

    if profil in ["ACCUEIL", "SECRETARIAT", "ADMIN"]:
        context["a_transmettre"] = qs.filter(
            sens="ENTRANT",
            statut="ENREGISTRE"
        ).count()

        context["retours_en_attente"] = RetourCourrier.objects.filter(
            traite=False,
            courrier__in=qs
        ).count()

    if profil == "RESPONSABLE":
        services = services_du_responsable(request.user)
        services_ids = services.values_list("id", flat=True)

        courriers_service = qs.filter(
            service_id__in=services_ids,
            sens="ENTRANT"
        )

        context["mes_services"] = services.count()

        context["a_traiter"] = courriers_service.filter(
            statut__in=["TRANSMIS", "EN_COURS", "EN_ATTENTE"]
        ).count()

        context["a_cloturer"] = courriers_service.filter(
            statut="TRAITE"
        ).count()

        context["retours_en_attente"] = RetourCourrier.objects.filter(
            traite=False,
            courrier__in=courriers_service
        ).count()

    if profil == "AGENT":
        affectations = Affectation.objects.filter(
            agent=request.user,
            active=True,
            courrier__in=qs
        )

        context["mes_affectations"] = affectations.count()

        context["a_traiter"] = affectations.filter(
            statut__in=["TRANSMIS", "EN_COURS", "EN_ATTENTE"]
        ).count()

        context["a_cloturer"] = affectations.filter(
            statut="TRAITE"
        ).count()

    return render(
        request,
        "courrier/dashboard.html",
        context
    )

@login_required
def liste_courriers(request):
    qs = courriers_accessibles(request.user)

    q = request.GET.get("q", "").strip()
    sens = request.GET.get("sens", "")
    statut = request.GET.get("statut", "")
    categorie = request.GET.get("categorie", "")

    if categorie == "entrants":
        qs = qs.filter(sens="ENTRANT")
        filtre_titre = "Courriers entrants"

    elif categorie == "sortants":
        qs = qs.filter(sens="SORTANT")
        filtre_titre = "Courriers sortants"

    elif categorie == "en_cours":
        qs = qs.filter(
            statut__in=["TRANSMIS", "EN_COURS", "EN_ATTENTE"]
        )
        filtre_titre = "Courriers en cours de traitement"

    elif categorie == "traites":
        qs = qs.filter(
            statut__in=["TRAITE", "CLOTURE"]
        )
        filtre_titre = "Courriers traités / clôturés"

    elif categorie == "en_retard":
        qs = qs.filter(
            date_limite__lt=timezone.localdate()
        ).exclude(
            statut__in=["TRAITE", "CLOTURE"]
        )
        filtre_titre = "Courriers en retard"

    else:
        filtre_titre = "Tous les courriers"

    if q:
        qs = qs.filter(
            Q(reference__icontains=q)
            | Q(objet__icontains=q)
            | Q(expediteur__icontains=q)
            | Q(destinataire__icontains=q)
        )

    if sens in ["ENTRANT", "SORTANT"]:
        qs = qs.filter(sens=sens)

    if statut:
        qs = qs.filter(statut=statut)

    nombre_par_page = request.GET.get("par_page", "10")

    if nombre_par_page not in ["10", "25", "50", "100"]:
        nombre_par_page = "10"

    paginator = Paginator(
        qs.order_by("-cree_le"),
        int(nombre_par_page)
    )

    page_obj = paginator.get_page(
        request.GET.get("page")
    )

    context = {
        "courriers": page_obj,
        "page_obj": page_obj,
        "paginator": paginator,
        "nombre_par_page": nombre_par_page,
        "q": q,
        "sens": sens,
        "statut": statut,
        "categorie": categorie,
        "filtre_titre": filtre_titre,
    }

    return render(
        request,
        "courrier/liste.html",
        context
    )

@login_required
def nouveau_courrier(request):
    if not utilisateur_peut_enregistrer(request.user):
        raise PermissionDenied(
            "Vous ne disposez pas du droit d'enregistrement."
        )

    if request.method == "POST":
        form = CourrierForm(request.POST, request.FILES)

        if form.is_valid():
            obj = form.save(commit=False)
            obj.cree_par = request.user

            if obj.sens == "ENTRANT":
                obj.service = None
                obj.responsable = None
                obj.date_expedition = None
                obj.statut = "ENREGISTRE"
                obj.date_cloture = None

            elif obj.sens == "SORTANT":
                if not obj.date_expedition:
                    form.add_error(
                        "date_expedition",
                        "La date d'expédition est obligatoire pour un courrier sortant."
                    )
                else:
                    obj.statut = "CLOTURE"
                    obj.date_cloture = obj.date_expedition
                    obj.service = None
                    obj.responsable = None

            if not form.errors:
                with transaction.atomic():
                    obj.save()
                    form.save_m2m()

                    documents = form.cleaned_data.get("documents", [])

                    for fichier in documents:
                        PieceJointe.objects.create(
                            courrier=obj,
                            fichier=fichier,
                            nom_original=fichier.name,
                            type_mime=getattr(fichier, "content_type", "") or "",
                            taille=fichier.size,
                            ajoute_par=request.user
                        )

                    enregistrer_historique(
                        obj,
                        request.user,
                        "ENREGISTREMENT",
                        details="Création du courrier.",
                        apres=(
                            f"Référence : {obj.reference}\n"
                            f"Sens : {obj.get_sens_display()}\n"
                            f"Objet : {obj.objet}"
                        )
                    )

                messages.success(
                    request,
                    f"Le courrier {obj.reference} a été enregistré avec succès."
                )

                return redirect("detail_courrier", pk=obj.pk)

    else:
        form = CourrierForm()

    return render(
        request,
        "courrier/formulaire.html",
        {
            "form": form,
            "titre": "Nouveau courrier",
            "libelle_bouton": "Enregistrer le courrier"
        }
    )

@login_required

        profil == "RESPONSABLE"

        and affectation_active.date_confirmation_reception is None
    )

@login_required
def transmettre_courrier(request, pk):
    obj = obtenir_courrier_accessible(request.user, pk)

    if not utilisateur_peut_transmettre(request.user, obj):
        raise PermissionDenied(
            "Vous ne disposez pas du droit de transmettre ce courrier."
        )

    if obj.sens != "ENTRANT" or obj.statut != "ENREGISTRE":
        raise PermissionDenied(
            "Seul un courrier entrant enregistré et non transmis peut être transmis."
        )

    if request.method == "POST":
        form = AffectationForm(request.POST)

        if form.is_valid():
            affectation = form.save(commit=False)

            if not affectation.service.actif:
                form.add_error(
                    "service",
                    "Ce service n'est plus actif."
                )

            if not affectation.service.responsable_id:
                form.add_error(
                    "service",
                    "Ce service ne possède pas de responsable."
                )

            if not form.errors:
                with transaction.atomic():
                    Affectation.objects.filter(
                        courrier=obj,
                        active=True
                    ).update(active=False)

                    affectation.courrier = obj
                    affectation.transmis_par = request.user
                    affectation.agent = None
                    affectation.active = True
                    affectation.save()

                    obj.service = affectation.service
                    obj.responsable = affectation.service.responsable
                    obj.statut = "TRANSMIS"

                    obj.save(
                        update_fields=[
                            "service",
                            "responsable",
                            "statut",
                            "modifie_le"
                        ]
                    )

                    enregistrer_historique(
                        obj,
                        request.user,
                        "TRANSMISSION",
                        details=(
                            f"Courrier transmis au service "
                            f"{affectation.service.nom}."
                        )
                    )

                messages.success(
                    request,
                    "Le courrier a été transmis au service sélectionné."
                )

                return redirect("detail_courrier", pk=obj.pk)

    else:
        form = AffectationForm()

    return render(
        request,
        "courrier/formulaire.html",
        {
            "form": form,
            "courrier": obj,
            "titre": "Transmettre le courrier",
            "libelle_bouton": "Confirmer la transmission"
        }
    )

            "Ce courrier n'est rattaché à aucun service."
        )

    if request.method == "POST":
        form = AffectationForm(request.POST)

        if form.is_valid():
            nouvelle = form.save(commit=False)

            if nouvelle.service_id != obj.service_id:
                form.add_error(
                    "service",
                    "L'affectation doit rester dans le service destinataire."
                )

            if not nouvelle.agent:
                form.add_error(
                    "agent",
                    "Veuillez sélectionner un agent."
                )

            if not form.errors:
                with transaction.atomic():
                    Affectation.objects.filter(
                        courrier=obj,
                        active=True
                    ).update(active=False)

                    nouvelle.courrier = obj
                    nouvelle.transmis_par = request.user
                    nouvelle.active = True
                    nouvelle.save()

                    enregistrer_historique(
                        obj,
                        request.user,
                        "AFFECTATION_AGENT",
                        details=(
                            f"Courrier affecté à "
                            f"{nouvelle.agent.get_full_name() or nouvelle.agent.username}."
                        )
                    )

                messages.success(
                    request,
                    "Le courrier a été affecté à l'agent."
                )

                return redirect("detail_courrier", pk=obj.pk)

    else:
        form = AffectationForm(
            initial={"service": obj.service_id}
        )

    return render(
        request,
        "courrier/formulaire.html",
        {
            "form": form,
            "courrier": obj,
            "titre": "Affecter à un agent",
            "libelle_bouton": "Confirmer l'affectation"
        }
    )

# FORMULAIRE DE CREATION

class CreationBordereauForm(forms.Form):

    service = forms.ModelChoiceField(
        queryset=Service.objects.none(),
        label="Service destinataire"
    )

    courriers = forms.ModelMultipleChoiceField(
        queryset=Courrier.objects.none(),
        widget=forms.CheckboxSelectMultiple,
        label="Courriers à transmettre"
    )

    instruction = forms.CharField(
        required=False,
        label="Instruction de transmission",
        widget=forms.Textarea(attrs={
            "rows": 3,
            "class": "form-control"
        })
    )

    def __init__(self, *args, user=None, **kwargs):
        super().__init__(*args, **kwargs)

        self.user = user

        self.fields["service"].queryset = (
            Service.objects
            .filter(actif=True)
            .select_related("responsable")
            .order_by("nom")
        )

        if user and obtenir_profil(user) == "RESPONSABLE":
            self.fields["service"].queryset = (
                Service.objects
                .filter(responsable=user, actif=True)
                .order_by("nom")
            )

        from .views import courriers_accessibles

        accessibles = courriers_accessibles(user)

        self.fields["courriers"].queryset = (
            accessibles
            .filter(
                sens=Courrier.Sens.ENTRANT,
                statut=Courrier.Statut.ENREGISTRE
            )
            .exclude(affectations__active=True)
            .distinct()
            .order_by("-cree_le")
        )

    def clean(self):
        cleaned = super().clean()

        service = cleaned.get("service")
        courriers = cleaned.get("courriers")

        if service and (
            not service.actif or not service.responsable_id
        ):
            self.add_error(
                "service",
                "Le service doit être actif et posséder un responsable."
            )

        if service and courriers:
            for courrier in courriers:

                if courrier.sens != Courrier.Sens.ENTRANT:
                    self.add_error(
                        "courriers",
                        f"Le courrier {courrier.reference} n'est pas entrant."
                    )

                if courrier.statut != Courrier.Statut.ENREGISTRE:
                    self.add_error(
                        "courriers",
                        f"Le courrier {courrier.reference} n'est plus disponible."
                    )

                if Affectation.objects.filter(
                    courrier=courrier,
                    active=True
                ).exists():
                    self.add_error(
                        "courriers",
                        f"Le courrier {courrier.reference} possède déjà une transmission active."
                    )

                if (
                    courrier.service_id
                    and courrier.service_id != service.pk
                ):
                    self.add_error(
                        "courriers",
                        f"Le courrier {courrier.reference} est rattaché à un autre service."
                    )

        return cleaned

@login_required
def creer_bordereau(request):

    profil = obtenir_profil(request.user)

    if profil not in ["ADMIN", "ACCUEIL", "SECRETARIAT"]:
        raise PermissionDenied

    if request.method == "POST":

        form = CreationBordereauForm(
            request.POST,
            user=request.user
        )

        if form.is_valid():

            service_selectionne = form.cleaned_data["service"]
            courriers_selectionnes = form.cleaned_data["courriers"]
            instruction = form.cleaned_data["instruction"]

            identifiants = list(
                courriers_selectionnes.values_list("pk", flat=True)
            )

            with transaction.atomic():

                service = (
                    Service.objects
                    .select_for_update()
                    .get(pk=service_selectionne.pk)
                )

                if not service.actif or not service.responsable_id:
                    raise PermissionDenied

                courriers_verrouilles = list(
                    Courrier.objects
                    .select_for_update()
                    .filter(pk__in=identifiants)
                    .order_by("pk")
                )

                if len(courriers_verrouilles) != len(identifiants):
                    raise PermissionDenied

                # Vérification de chaque courrier avant transmission.
                for courrier in courriers_verrouilles:

                    if not utilisateur_peut_transmettre(
                        request.user,
                        courrier
                    ):
                        raise PermissionDenied

                    if courrier.sens != Courrier.Sens.ENTRANT:
                        raise PermissionDenied

                    if courrier.statut != Courrier.Statut.ENREGISTRE:
                        raise PermissionDenied

                    if Affectation.objects.filter(
                        courrier=courrier,
                        active=True
                    ).exists():
                        raise PermissionDenied

                    if (
                        courrier.service_id
                        and courrier.service_id != service.pk
                    ):
                        raise PermissionDenied

                # Création du bordereau.
                bordereau = BordereauTransmission.objects.create(
                    service=service,
                    transmis_par=request.user
                )

                for courrier in courriers_verrouilles:

                    LigneBordereau.objects.create(
                        bordereau=bordereau,
                        courrier=courrier
                    )

                    Affectation.objects.create(
                        courrier=courrier,
                        service=service,
                        agent=None,
                        transmis_par=request.user,
                        date_transmission=timezone.now(),
                        instruction=instruction,
                        active=True,
                        statut="TRANSMIS"
                    )

                    courrier.service = service
                    courrier.responsable = service.responsable
                    courrier.statut = Courrier.Statut.TRANSMIS

                    courrier.save(
                        update_fields=[
                            "service",
                            "responsable",
                            "statut",
                            "modifie_le"
                        ]
                    )

                    enregistrer_historique_bordereau(
                        courrier,
                        request.user,
                        "TRANSMISSION_BORDEREAU",
                        (
                            f"Courrier transmis au service {service.nom} "
                            f"par le bordereau numéro {bordereau.pk}."
                        )
                    )

            messages.success(
                request,
                f"Bordereau numéro {bordereau.pk} créé avec succès."
            )

            return redirect(
                "detail_bordereau",
                pk=bordereau.pk
            )

    else:

        form = CreationBordereauForm(
            user=request.user
        )

    return render(
        request,
        "courrier/bordereaux/formulaire.html",
        {
            "form": form
        }
    )

        BordereauTransmission.objects.select_related(

        profil in [

            "ACCUEIL",
            "SECRETARIAT"
        ]

            profil == "RESPONSABLE"
            and utilisateur_est_responsable_du_service(

        .order_by("pk")

        profil == "RESPONSABLE"

            request.user,
            bordereau.service
        )
        and bordereau.statut == "TRANSMIS"
    )

        bordereau,
        "reception",

    )

    if request.method != "POST":

    profil = obtenir_profil(request.user)

    bordereau = get_object_or_404(
        BordereauTransmission.objects.select_related("service"),
        pk=pk
    )

    if (
        profil != "RESPONSABLE"
        or not utilisateur_est_responsable_du_service(
            request.user,
            bordereau.service
        )
    ):
        raise PermissionDenied

    form = ReceptionBordereauForm(request.POST)

    if not form.is_valid():

        messages.error(
            request,
            "Les informations de réception sont invalides."
        )

        return redirect(
            "detail_bordereau",
            pk=pk
        )

    with transaction.atomic():

        bordereau = (
            BordereauTransmission.objects
            .select_for_update()
            .select_related("service")
            .get(pk=pk)
        )

        if not utilisateur_est_responsable_du_service(
            request.user,
            bordereau.service
        ):
            raise PermissionDenied

        if bordereau.statut != "TRANSMIS":

            messages.warning(
                request,
                "Ce bordereau a déjà été réceptionné."
            )

            return redirect(
                "detail_bordereau",
                pk=pk
            )

        lignes = list(
            LigneBordereau.objects
            .filter(bordereau=bordereau)
            .select_related("courrier")
            .order_by("pk")
        )

        if not lignes:
            messages.error(
                request,
                "Ce bordereau ne contient aucun courrier."
            )

            return redirect(
                "detail_bordereau",
                pk=pk
            )

        ReceptionBordereau.objects.create(
            bordereau=bordereau,
            service=bordereau.service,
            receptionne_par=request.user,
            observation=form.cleaned_data["observation"]
        )

        bordereau.statut = "RECEPTIONNE"

        bordereau.save(
            update_fields=["statut"]
        )

        for ligne in lignes:

            affectation = (
                Affectation.objects
                .select_for_update()
                .filter(
                    courrier=ligne.courrier,
                    service=bordereau.service,
                    active=True
                )
                .order_by("-date_transmission", "-pk")
                .first()
            )

            if affectation and not affectation.date_confirmation_reception:

                affectation.date_confirmation_reception = timezone.now()
                affectation.confirme_par = request.user

                affectation.save(
                    update_fields=[
                        "date_confirmation_reception",
                        "confirme_par"
                    ]
                )

            enregistrer_historique_bordereau(
                ligne.courrier,
                request.user,
                "RECEPTION_BORDEREAU",
                (
                    f"Réception confirmée par {request.user} "
                    f"pour le bordereau numéro {bordereau.pk}."
                )
            )

    messages.success(
        request,
        "La réception du bordereau et de ses courriers a été enregistrée."
    )

    return redirect(
        "detail_bordereau",
        pk=pk
    )

def obtenir_courriers_export(request):
    """
    Récupère uniquement les courriers accessibles
    à l'utilisateur connecté et applique les filtres.
    """

    courriers = courriers_accessibles(request.user)

    recherche = request.GET.get("q", "").strip()
    sens = request.GET.get("sens", "").strip()
    statut = request.GET.get("statut", "").strip()

    if recherche:
        courriers = courriers.filter(
            Q(reference__icontains=recherche)
            | Q(objet__icontains=recherche)
            | Q(expediteur__icontains=recherche)
            | Q(destinataire__icontains=recherche)
            | Q(description__icontains=recherche)
        )

    if sens in [
        Courrier.Sens.ENTRANT,
        Courrier.Sens.SORTANT,
    ]:
        courriers = courriers.filter(sens=sens)

    statuts_valides = [
        choix[0] for choix in Courrier.Statut.choices
    ]

    if statut in statuts_valides:
        courriers = courriers.filter(statut=statut)

    return courriers.select_related(
        "service",
        "responsable",
        "categorie",
        "cree_par",
    ).order_by("-date_courrier", "-pk")


def valeur_texte(valeur):
    if valeur is None:
        return ""

    return str(valeur)

        "CelluleGSC",
        parent=styles["Normal"],
        fontName="Helvetica",
        fontSize=6.5,

        alignment=TA_LEFT,
    )

        textColor=colors.white,
        alignment=TA_CENTER,

        "Date",
        "Sens",
        "Expéditeur",
        "Destinataire",
        "Objet",
        "Service",
        "Statut",
    ]

    donnees = [
        [Paragraph(titre, entete_style) for titre in entetes]

    for courrier in courriers.iterator():

    )

    return response
