# SERVICE / DIRECTION / DEPARTEMENT / UNITE
# =====================================================

class Service(models.Model):

    class TypeStructure(models.TextChoices):
        DIRECTION = "DIRECTION", "Direction"
        DEPARTEMENT = "DEPARTEMENT", "Département"
        SERVICE = "SERVICE", "Service"
        UNITE = "UNITE", "Unité"

    nom = models.CharField(
        max_length=120,
        unique=True,
        verbose_name="Nom de la structure"
    )

    description = models.TextField(
        blank=True,
        verbose_name="Description"
    )

    # Ancien champ conservé pour compatibilité.
    responsable = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="services_diriges",
        null=True,
        blank=True,
        verbose_name="Responsable principal"
    )

    # Nouvelle organisation hiérarchique.
    type_structure = models.CharField(
        max_length=20,
        choices=TypeStructure.choices,
        default=TypeStructure.SERVICE,
        verbose_name="Type de structure"
    )

    parent = models.ForeignKey(
        "self",
        on_delete=models.PROTECT,
        related_name="structures_enfants",
        null=True,
        blank=True,
        verbose_name="Structure de rattachement"
    )

    # Plusieurs responsables possibles.
    responsables = models.ManyToManyField(
        settings.AUTH_USER_MODEL,
        through="ResponsableService",
        through_fields=("service", "utilisateur"),
        related_name="services_responsable_multi",
        blank=True,
        verbose_name="Responsables"
    )

    actif = models.BooleanField(
        default=True,
        verbose_name="Structure active"
    )

    class Meta:
        ordering = ["nom"]
        verbose_name = "Structure"
        verbose_name_plural = "Structures"

    def __str__(self):
        return self.nom

        STRICTE = "STRICTE", "Strictement confidentielle"

    class Statut(models.TextChoices):
        ENREGISTRE = "ENREGISTRE", "Enregistré"

    mode_reception = models.CharField(
        max_length=20,

        null=True,
        blank=True

    nom_livreur = models.CharField(
        max_length=180,
        blank=True

    date_expedition = models.DateField(
        null=True,
        blank=True
    )

    expediteur = models.CharField(max_length=180)
    destinataire = models.CharField(max_length=180)
    objet = models.CharField(max_length=250)
    description = models.TextField(blank=True)

    categorie = models.ForeignKey(

    priorite = models.CharField(

        choices=Priorite.choices,
        default=Priorite.NORMALE
    )

    confidentialite = models.CharField(
        max_length=20,
        choices=Confidentialite.choices,
        default=Confidentialite.NORMALE
    )

    statut = models.CharField(
        max_length=20,
        choices=Statut.choices,
        default=Statut.ENREGISTRE
    )

    # Informations du moteur d'orientation.
    orientation_statut = models.CharField(
        max_length=20,
        choices=StatutOrientation.choices,
        default=StatutOrientation.NON_EVALUE
    )

    orientation_confiance = models.DecimalField(
        max_digits=5,
        decimal_places=2,
        null=True,
        blank=True
    )

    orientation_motif = models.TextField(blank=True)

    orientation_service_propose = models.ForeignKey(
        Service,
        on_delete=models.PROTECT,
        related_name="orientations_proposees",
        null=True,
        blank=True
    )

    orientation_validee_par = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="orientations_validees",
        null=True,
        blank=True
    )

    orientation_date = models.DateTimeField(
        null=True,
        blank=True
    )

    orientation_automatique = models.BooleanField(default=False)

    date_limite = models.DateField(
        null=True,
        blank=True
    )

    date_cloture = models.DateField(
        null=True,
        blank=True
    )

    document = models.FileField(
        upload_to="courriers/%Y/%m/",
        blank=True,
        null=True
    )

    courrier_origine = models.ForeignKey(
        "self",
        on_delete=models.PROTECT,
        related_name="courriers_reponse",
        null=True,
        blank=True
    )

    cree_par = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="courriers_crees"
    )

    cree_le = models.DateTimeField(auto_now_add=True)
    modifie_le = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-cree_le"]
        verbose_name = "Courrier"
        verbose_name_plural = "Courriers"

        permissions = [
            ("transmettre_courrier", "Peut transmettre un courrier"),
            ("cloturer_courrier", "Peut clôturer un courrier"),
            ("traiter_courrier", "Peut traiter un courrier"),
            ("retourner_courrier", "Peut retourner un courrier"),
            ("reouvrir_courrier", "Peut rouvrir un courrier"),
            ("valider_orientation", "Peut valider une orientation"),
            ("consulter_confidentiel", "Peut consulter un courrier confidentiel"),
        ]

    def save(self, *args, **kwargs):

        if not self.reference:

            prefix = (
                "ENT"
                if self.sens == self.Sens.ENTRANT
                else "SOR"
            )

            annee = timezone.localdate().year

            with transaction.atomic():

                compteur, cree = (
                    CompteurReference.objects
                    .select_for_update()
                    .get_or_create(
                        sens=self.sens,
                        annee=annee,
                        defaults={"dernier_numero": 0}
                    )
                )

                if cree or compteur.dernier_numero == 0:

                    anciennes_references = (
                        Courrier.objects
                        .filter(
                            reference__startswith=f"{prefix}-{annee}-"
                        )
                        .values_list("reference", flat=True)
                    )

                    maximum = 0

                    for ancienne in anciennes_references:
                        try:
                            numero = int(ancienne.rsplit("-", 1)[1])
                            maximum = max(maximum, numero)
                        except (ValueError, IndexError):
                            continue

                    compteur.dernier_numero = max(
                        compteur.dernier_numero,
                        maximum
                    )

                compteur.dernier_numero += 1

                compteur.save(
                    update_fields=["dernier_numero"]
                )

                self.reference = (
                    f"{prefix}-{annee}-"
                    f"{compteur.dernier_numero:06d}"
                )

                super().save(*args, **kwargs)

        else:
            super().save(*args, **kwargs)

    def est_strictement_confidentiel(self):
        return self.confidentialite == self.Confidentialite.STRICTE

    def __str__(self):
        return f"{self.reference} – {self.objet}"

class Affectation(models.Model):

    class TypeAffectation(models.TextChoices):
        SERVICE = "SERVICE", "Transmission au service"
        AGENT = "AGENT", "Affectation à un agent"

    courrier = models.ForeignKey(
        Courrier,
        on_delete=models.PROTECT,
        related_name="affectations"
    )

    service = models.ForeignKey(
        Service,
        on_delete=models.PROTECT,
        related_name="affectations"
    )

    agent = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        null=True,
        blank=True,
        related_name="affectations_agent"
    )

    type_affectation = models.CharField(
        max_length=20,
        choices=TypeAffectation.choices,
        default=TypeAffectation.SERVICE
    )

    transmis_par = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="transmissions_effectuees"
    )

    date_transmission = models.DateTimeField(
        default=timezone.now
    )

    instruction = models.TextField(blank=True)

    date_limite = models.DateField(
        null=True,
        blank=True
    )

    statut = models.CharField(
        max_length=30,
        default="TRANSMIS"
    )

    observation = models.TextField(blank=True)

    active = models.BooleanField(
        default=True,
        verbose_name="Affectation active"
    )

    date_confirmation_reception = models.DateTimeField(
        null=True,
        blank=True
    )

    confirme_par = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="affectations_confirmees",
        null=True,
        blank=True
    )

    class Meta:
        ordering = ["-date_transmission"]
        verbose_name = "Affectation"
        verbose_name_plural = "Affectations"

    def save(self, *args, **kwargs):

        # Déduction du type pour les anciennes créations.
        if self.agent_id and self.type_affectation == self.TypeAffectation.SERVICE:
            self.type_affectation = self.TypeAffectation.AGENT

        super().save(*args, **kwargs)

    def __str__(self):
        if self.agent_id:
            return f"{self.courrier.reference} - {self.agent}"
        return f"{self.courrier.reference} - {self.service}"

    courrier = models.ForeignKey(
        Courrier,
        on_delete=models.PROTECT,

    utilisateur = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="autorisations_confidentialite"
    )

    type_acces = models.CharField(
        max_length=20,
        choices=TypeAcces.choices,
        default=TypeAcces.CONTENU
    )

    autorise_par = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="autorisations_confidentialite_accordees"
    )

    motif = models.TextField()

    date_debut = models.DateTimeField(default=timezone.now)

    date_fin = models.DateTimeField(
        null=True,
        blank=True
    )

    actif = models.BooleanField(default=True)

    revoque_par = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.PROTECT,
        related_name="autorisations_confidentialite_revoquees",
        null=True,
        blank=True
    )

    date_revocation = models.DateTimeField(
        null=True,
        blank=True
    )

    date_creation = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["-date_creation"]
        verbose_name = "Autorisation confidentielle"
        verbose_name_plural = "Autorisations confidentielles"

    def est_valide(self):
        maintenant = timezone.now()

        return (
            self.actif
            and self.date_debut <= maintenant
            and (
                self.date_fin is None
                or self.date_fin >= maintenant
            )
            and self.date_revocation is None
        )

        return (
            f"{self.utilisateur} - "

def normaliser_nom(valeur):
    valeur = unicodedata.normalize("NFD", str(valeur or ""))
    valeur = "".join(
        caractere
        for caractere in valeur
        if unicodedata.category(caractere) != "Mn"
    )
    return " ".join(valeur.strip().casefold().split())

# ============================================================

def obtenir_profil(user):
    if (
        user is None
        or not getattr(user, "is_authenticated", False)
        or not getattr(user, "is_active", False)
    ):
        return None

    # L'administrateur technique n'a aucun accès confidentiel
    # automatique. Les autorisations explicites restent applicables.
    if getattr(user, "is_superuser", False):
        return "ADMIN"

    groupes = {
        normaliser_nom(nom)
        for nom in user.groups.values_list("name", flat=True)
    }

    correspondances = [
        ("accueil", "ACCUEIL"),
        ("secretariat", "SECRETARIAT"),
        ("direction", "DIRECTION"),
        ("responsable de service", "RESPONSABLE"),
        ("responsable", "RESPONSABLE"),
        ("agent", "AGENT"),
    ]

    for nom, profil in correspondances:
        if normaliser_nom(nom) in groupes:
            return profil

    return None


def obtenir_profils(user):
    profil = obtenir_profil(user)
    return {profil} if profil else set()

def responsabilites_actives(user):
    if not user or not getattr(user, "is_authenticated", False):
        return ResponsableService.objects.none()

    return ResponsableService.objects.filter(
        utilisateur=user,
        actif=True,
        service__actif=True,
    ).select_related(
        "service",
        "service__parent",
    ).prefetch_related(
        "categories",
    )


def services_du_responsable(user):
    if obtenir_profil(user) != "RESPONSABLE":
        return Service.objects.none()

    services_configures = Service.objects.filter(
        responsables_configures__utilisateur=user,
        responsables_configures__actif=True,
        actif=True,
    )

    # Compatibilité avec l'ancien champ Service.responsable.
    anciens_services = Service.objects.filter(
        responsable=user,
        actif=True,
    )

    return (services_configures | anciens_services).distinct()


def utilisateur_est_responsable_du_service(user, service):
    if (
        obtenir_profil(user) not in ["RESPONSABLE", "DIRECTION"]
        or service is None
        or not service.actif
    ):
        return False

    if service.responsable_id == user.pk:
        return True

    return ResponsableService.objects.filter(
        utilisateur=user,
        service=service,
        actif=True,
        service__actif=True,
    ).exists()


def responsabilite_sur_service(user, service):
    if service is None:
        return None

    return ResponsableService.objects.filter(
        utilisateur=user,
        service=service,
        actif=True,
        service__actif=True,
    ).first()


def services_accessibles_du_responsable(user):
    services = services_du_responsable(user)
    ids = set(services.values_list("id", flat=True))

    responsabilites = responsabilites_actives(user).filter(
        inclure_structures_enfants=True
    )

    for responsabilite in responsabilites:
        parent = responsabilite.service

        nouveaux_ids = set(
            Service.objects.filter(
                parent=parent,
                actif=True,
            ).values_list("id", flat=True)
        )

        ids.update(nouveaux_ids)

        # Recherche récursive des structures descendantes.
        while nouveaux_ids:
            enfants = Service.objects.filter(
                parent_id__in=nouveaux_ids,
                actif=True,
            )

            nouveaux_ids = set(
                enfants.values_list("id", flat=True)
            )

            ids.update(nouveaux_ids)

    return Service.objects.filter(
        id__in=ids,
        actif=True,
    )


def utilisateur_est_responsable_du_courrier(user, courrier):
    if courrier is None or not courrier.service_id:
        return False

    if obtenir_profil(user) != "RESPONSABLE":
        return False

    return services_accessibles_du_responsable(user).filter(
        pk=courrier.service_id
    ).exists()


def responsable_peut_gerer_categorie(user, courrier):
    if courrier is None or not courrier.service_id:
        return False

    responsabilites = responsabilites_actives(user).filter(
        service_id=courrier.service_id
    )

    if Service.objects.filter(
        pk=courrier.service_id,
        responsable=user,
    ).exists():
        return True

    for responsabilite in responsabilites:
        if responsabilite.toutes_categories:
            return True

        if not courrier.categorie_id:
            continue

        if responsabilite.categories.filter(
            pk=courrier.categorie_id
        ).exists():
            return True

    return False

    responsabilite = responsabilite_sur_service(user, service)

    if responsabilite:
        return responsabilite.peut_affecter_agents

    return service.responsable_id == getattr(user, "pk", None)


def responsable_peut_cloturer(user, service):

    responsabilite = responsabilite_sur_service(user, service)

    if responsabilite:

def courrier_est_confidentiel(courrier):
    if courrier is None:
        return False

    return (
        courrier.confidentialite in [
            Courrier.Confidentialite.CONFIDENTIELLE,
            Courrier.Confidentialite.STRICTE,
        ]
        or courrier.priorite == Courrier.Priorite.CONFIDENTIELLE

def courrier_est_strictement_confidentiel(courrier):
    if courrier is None:
        return False

    return (
        courrier.confidentialite
        == Courrier.Confidentialite.STRICTE
    )


def autorisation_confidentielle_valide(
    user,
    courrier,
    type_acces="CONTENU",
):
    if (
        user is None
        or courrier is None
        or not getattr(user, "is_authenticated", False)
        or not getattr(user, "is_active", False)
    ):
        return False

    maintenant = timezone.now()

    return AutorisationConfidentialite.objects.filter(
        utilisateur=user,
        courrier=courrier,
        type_acces=type_acces,
        actif=True,
        date_debut__lte=maintenant,
        date_revocation__isnull=True,
    ).filter(
        Q(date_fin__isnull=True)
        | Q(date_fin__gte=maintenant)
    ).exists()


def utilisateur_est_destinataire_confidentiel(user, courrier):
    return bool(
        user
        and courrier
        and getattr(user, "is_authenticated", False)
        and getattr(user, "is_active", False)
        and courrier.destinataire_confidentiel_id == user.pk
    )


def utilisateur_peut_consulter_contenu_confidentiel(

    if (

        or courrier is None
        or not getattr(user, "is_authenticated", False)

    if not courrier_est_confidentiel(courrier):
        return utilisateur_peut_consulter_courrier(user, courrier)

    # Le destinataire expressément désigné est autorisé.
    if utilisateur_est_destinataire_confidentiel(user, courrier):
        return True

    # Sinon, une autorisation CONTENU valide est obligatoire.
    return autorisation_confidentielle_valide(
        user,
        courrier,
        "CONTENU",
    )

# ============================================================
# CONSULTATION DES METADONNEES
# ============================================================

def utilisateur_peut_consulter_courrier(user, courrier):
    profil = obtenir_profil(user)

    if profil is None or courrier is None:
        return False

    # Pour les courriers confidentiels, le destinataire ou une
    # personne expressément autorisée peut accéder à la fiche.
    # Cela ne donne pas automatiquement accès aux pièces jointes
    # ni à l'historique : ces droits sont contrôlés séparément.
    if courrier_est_confidentiel(courrier):
        if utilisateur_est_destinataire_confidentiel(user, courrier):
            return True

        autorisations = (
            autorisation_confidentielle_valide(
                user, courrier, "CONTENU"
            )
            or autorisation_confidentielle_valide(
                user, courrier, "PIECES_JOINTES"
            )
            or autorisation_confidentielle_valide(
                user, courrier, "HISTORIQUE"
            )
        )

        if autorisations:
            return True

    if not utilisateur_peut_consulter(user):
        return False

    # L'administrateur technique consulte les courriers ordinaires.
    # Aucun accès confidentiel automatique.
    if profil == "ADMIN":
        return not courrier_est_confidentiel(courrier)

    # Accueil et Secrétariat : métadonnées administratives.
    if profil in ["ACCUEIL", "SECRETARIAT"]:
        return True

    # Direction : uniquement sa propre structure.
    if profil == "DIRECTION":
        if not courrier.service_id:
            return False

        return utilisateur_est_responsable_du_service(
            user,
            courrier.service,
        )

    # Responsable : périmètre de responsabilité et catégories.
    if profil == "RESPONSABLE":
        if not utilisateur_est_responsable_du_courrier(
            user,
            courrier,
        ):
            return False

        return responsable_peut_gerer_categorie(
            user,
            courrier,
        )

    # Agent : affectation active obligatoire.
    if profil == "AGENT":
        return courrier.affectations.filter(
            agent=user,
            active=True,
        ).exists()

    return False


# ============================================================
# CONSULTATION DU CONTENU
# ============================================================

def utilisateur_peut_consulter_contenu_courrier(
    user,
    courrier,
):
    if user is None or courrier is None:
        return False

    # Le contrôle du contenu confidentiel est indépendant du
    # périmètre ordinaire de consultation des métadonnées.
    if courrier_est_confidentiel(courrier):
        return utilisateur_peut_consulter_contenu_confidentiel(
            user,
            courrier,
        )

    return (
        utilisateur_peut_consulter_courrier(user, courrier)
        and getattr(user, "is_active", False)
    )


# ============================================================
# MODIFICATION D'UN COURRIER
# ============================================================

def utilisateur_peut_modifier_courrier(user, courrier):
    profil = obtenir_profil(user)

    if profil is None or courrier is None:
        return False

    if profil in ["ACCUEIL", "DIRECTION", "AGENT"]:
        return False

    if not user.has_perm("courrier.change_courrier"):
        return False

    # L'administrateur ne modifie pas automatiquement les courriers
    # confidentiels.
    if profil == "ADMIN":
        return not courrier_est_confidentiel(courrier)

    if not utilisateur_peut_consulter_contenu_courrier(
        user,
        courrier,
    ):
        return False

    if profil == "SECRETARIAT":
        return not courrier_est_deja_transmis(courrier)

    if profil == "RESPONSABLE":
        return (
            utilisateur_est_responsable_du_courrier(
                user,
                courrier,
            )
            and courrier.statut != Courrier.Statut.CLOTURE
        )

    return False


# ============================================================
# TRANSMISSION INITIALE
# ============================================================

def utilisateur_peut_transmettre(user, courrier=None):
    profil = obtenir_profil(user)

    if profil not in ["ACCUEIL", "SECRETARIAT"]:
        return False

    if not user.has_perm("courrier.transmettre_courrier"):
        return False

    if courrier is not None:
        if courrier.sens != Courrier.Sens.ENTRANT:
            return False

        if courrier.statut != Courrier.Statut.ENREGISTRE:
            return False

        if courrier_est_deja_transmis(courrier):
            return False

    if profil == "ACCUEIL":
        return True

    return utilisateur_a_delegation_valide(user)


# ============================================================
# AFFECTATION A UN AGENT
# ============================================================

def utilisateur_peut_affecter_agent(user, courrier):
    if obtenir_profil(user) != "RESPONSABLE":
        return False

    if courrier is None or not courrier.service_id:
        return False

    if not user.has_perm("courrier.change_courrier"):
        return False

    if not utilisateur_est_responsable_du_courrier(
        user,
        courrier,
    ):
        return False

    if not responsable_peut_affecter_agents(
        user,
        courrier.service,
    ):
        return False

    if not utilisateur_peut_consulter_contenu_courrier(
        user,
        courrier,
    ):
        return False

    return (
        courrier.sens == Courrier.Sens.ENTRANT
        and courrier.statut in [
            Courrier.Statut.TRANSMIS,
            Courrier.Statut.EN_COURS,
            Courrier.Statut.EN_ATTENTE,
        ]
    )


# ============================================================
# VERIFICATION D'UNE AFFECTATION AGENT
# ============================================================

def agent_est_affecte_au_courrier(user, courrier):
    if user is None or courrier is None:
        return False

    return Affectation.objects.filter(
        courrier=courrier,
        agent=user,
        active=True,
    ).exists()


# ============================================================
# TRAITEMENT
# ============================================================

def utilisateur_peut_traiter_courrier(user, courrier):
    profil = obtenir_profil(user)

    if profil not in ["AGENT", "RESPONSABLE"]:
        return False

    if courrier is None:
        return False

    if courrier.sens != Courrier.Sens.ENTRANT:
        return False

    if courrier.statut not in [
        Courrier.Statut.TRANSMIS,
        Courrier.Statut.EN_COURS,
        Courrier.Statut.EN_ATTENTE,
    ]:
        return False

    if not user.has_perm("courrier.traiter_courrier"):
        return False

    if not utilisateur_peut_consulter_contenu_courrier(
        user,
        courrier,
    ):
        return False

    if profil == "AGENT":
        return agent_est_affecte_au_courrier(
            user,
            courrier,
        )

    return (
        utilisateur_est_responsable_du_courrier(
            user,
            courrier,
        )
        and responsable_peut_gerer_categorie(
            user,
            courrier,
        )
    )


# ============================================================
# CLOTURE
# ============================================================

def utilisateur_peut_cloturer(user):
    return (
        obtenir_profil(user) == "RESPONSABLE"
        and user.has_perm("courrier.cloturer_courrier")
    )


def utilisateur_peut_cloturer_courrier(user, courrier):
    if obtenir_profil(user) != "RESPONSABLE":
        return False

    if courrier is None or not courrier.service_id:
        return False

    if not user.has_perm("courrier.cloturer_courrier"):
        return False

    if not utilisateur_peut_consulter_contenu_courrier(
        user,
        courrier,
    ):
        return False

    return (
        courrier.sens == Courrier.Sens.ENTRANT
        and courrier.statut == Courrier.Statut.TRAITE
        and utilisateur_est_responsable_du_courrier(
            user,
            courrier,
        )
        and responsable_peut_cloturer(
            user,
            courrier.service,
        )
    )


# ============================================================
# RETOUR AU SECRETARIAT
# ============================================================

def utilisateur_peut_retourner_courrier(user, courrier):
    if obtenir_profil(user) != "RESPONSABLE":
        return False

    if courrier is None:
        return False

    if not user.has_perm("courrier.retourner_courrier"):
        return False

    if not utilisateur_peut_consulter_contenu_courrier(
        user,
        courrier,
    ):
        return False

    return (
        courrier.sens == Courrier.Sens.ENTRANT
        and courrier.statut in [
            Courrier.Statut.TRANSMIS,
            Courrier.Statut.EN_COURS,
            Courrier.Statut.EN_ATTENTE,
            Courrier.Statut.TRAITE,
        ]
        and utilisateur_est_responsable_du_courrier(
            user,
            courrier,
        )
    )


# ============================================================
# REOUVERTURE
# ============================================================

def utilisateur_peut_reouvrir_courrier(user, courrier):
    profil = obtenir_profil(user)

    if profil is None or courrier is None:
        return False

    if courrier.statut != Courrier.Statut.CLOTURE:
        return False

    if not user.has_perm("courrier.reouvrir_courrier"):
        return False

    if not utilisateur_peut_consulter_contenu_courrier(
        user,
        courrier,
    ):
        return False

    if profil == "ADMIN":
        return not courrier_est_confidentiel(courrier)

    if profil == "RESPONSABLE":
        return (
            courrier.sens == Courrier.Sens.ENTRANT
            and utilisateur_est_responsable_du_courrier(
                user,
                courrier,
            )
        )

    return False


# ============================================================
# VALIDATION D'ORIENTATION
# ============================================================

def utilisateur_peut_valider_orientation(user):
    profil = obtenir_profil(user)

    return (
        profil in ["ACCUEIL", "SECRETARIAT"]
        and user.has_perm("courrier.valider_orientation")
    )


# ============================================================
# GESTION DES AUTORISATIONS CONFIDENTIELLES
# ============================================================

def utilisateur_peut_gerer_autorisations_confidentielles(
    user,
    courrier,
):
    if user is None or courrier is None:
        return False

    profil = obtenir_profil(user)

    if profil not in ["SECRETARIAT", "RESPONSABLE", "DIRECTION"]:
        return False

    if not user.has_perm("courrier.consulter_confidentiel"):
        return False

    if profil == "SECRETARIAT":
        return True

    if profil == "DIRECTION":
        return (
            courrier.service_id is not None
            and utilisateur_est_responsable_du_service(
                user,
                courrier.service,
            )
        )

    return utilisateur_est_responsable_du_courrier(
        user,
        courrier,
    )
